===== APPENDIX A: stage7_funding_enum.py =====
#!/usr/bin/env python3
# Stage 7 independent re-computation of funding-settlement event counts (analysis only; no engine/evaluator import).
# Uses explicit integer-second timestamp SETS (not formulas) for settlement times over a 10-day horizon starting 2026-01-05T00:00Z (arbitrary Monday).
import itertools, sys
H=3600; BAR=4*H; DAY=24*H
T0=0                      # 2026-01-05T00:00:00Z as relative second 0
HORIZON=10*DAY
fails=[]
def check(c,msg):
    print(("PASS " if c else "FAIL ")+msg)
    if not c: fails.append(msg)
def stamps(interval_h, phase_h=0):
    return {t for t in range(phase_h*H, HORIZON, interval_h*H)}
def count(S, entry, exit_, e_incl, x_incl):
    n=0
    for t in S:
        a = t>=entry if e_incl else t>entry
        b = t<=exit_ if x_incl else t<exit_
        if a and b: n+=1
    return n
CONV=[("in/in",True,True),("in/ex",True,False),("ex/in",False,True),("ex/ex",False,False)]
check(30*BAR==120*H,"30 bars x 4h = 120 hours")
EXPECT={ (8,"aligned"):(16,15,15,14),(8,"unaligned"):(15,15,15,15),(4,"any"):(31,30,30,29),(1,"any"):(121,120,120,119)}
for ih in (8,4,1):
    S=stamps(ih)
    for hour in (0,4,8,12,16,20):
        entry=2*DAY+hour*H; exit_=entry+30*BAR
        got=tuple(count(S,entry,exit_,e,x) for _,e,x in CONV)
        if ih==8: key=(8,"aligned" if hour%8==0 else "unaligned")
        else: key=(ih,"any")
        check(got==EXPECT[key],"interval %dh entry hour %02d: counts in/in,in/ex,ex/in,ex/ex = %s expected %s"%(ih,hour,got,EXPECT[key]))
# early SL/TP exits: exit inside bar k (k bars started, exit after f seconds of the last bar)
print("\nEarly exits, 8h interval, entry 00:00 (aligned) and 04:00, convention in/ex (entry counts, exit instant does not):")
S8=stamps(8)
for hour in (0,4):
    entry=2*DAY+hour*H
    for k,f in ((1,60),(2,BAR-1),(3,1),(6,BAR//2),(15,BAR-1),(30,BAR//2),(30,BAR)):
        exit_=entry+(k-1)*BAR+f
        print("  entry %02d:00 exit after bar %d +%ds: events %d"%(hour,k,f,count(S8,entry,exit_,True,False)))
# a position closed immediately before / at / after a settlement
t=2*DAY+8*H
for lab,ex in (("1s before",t-1),("exactly at",t),("1s after",t+1)):
    c=[count(S8,2*DAY,ex,True,x) for x in (False,True)]
    print("  entry 2026-01-07T00:00 closed %s the 08:00 settlement: events (exit excl/incl) = %s"%(lab,c))
check(count(S8,2*DAY,t-1,True,False)==1 and count(S8,2*DAY,t+1,True,False)==2,"closed 1s before 08:00 settlement pays only the entry-instant event; 1s after pays 2 (00:00 entry-inclusive + 08:00)")
# rate normalisation: r8 per 8h expressed per event for 4h and 1h interval (time-proportional assumption, NOT data)
print("\nRate normalisation (time-proportional equivalence; an assumption, not funding history):")
r8=0.0010
r4,r1=r8*4/8,r8*1/8
check(abs(r4-0.0005)<1e-12 and abs(r1-0.000125)<1e-12,"0.10%%/8h = %.4f%%/4h = %.5f%%/1h"%(r4*100,r1*100))
check(abs(15*r8-30*r4)<1e-12 and abs(15*r8-120*r1)<1e-12,"total over 120h identical: 15x0.10%% = 30x0.05%% = 120x0.0125%% = 1.50%%")
print("\nOne-event R impact (event rate / risk%):")
for rp in (0.015,0.03):
    for lab,r in (("0.10%/8h",r8),("0.05%/4h",r4),("0.0125%/1h",r1),("0.01%/8h",0.0001)):
        print("  risk %.1f%%  event rate %-10s -> %.4f R"%(rp*100,lab,r/rp))
check(abs(r8/0.015-0.0667)<1e-3 and abs(r8/0.03-0.0333)<1e-3,"one 8h event at 0.10%: 0.0667R (1.5% risk), 0.0333R (3%)")
check(abs(r4/0.015-0.0333)<1e-3 and abs(r4/0.03-0.0167)<1e-3,"one 4h event at 0.05%: 0.0333R (1.5%), 0.0167R (3%)")
# Stage 5/6 table spot checks: 30 bars, 15 events
for rp,expL in ((0.015,-1.0),(0.03,-0.5)):
    check(abs(-15*r8/rp-expL)<1e-9,"L real funding +0.10%%/8h, 30 bars, risk %.1f%%: %.2fR"%(rp*100,expL))
check(abs(-0.00005*30/0.015+0.10)<1e-9 and abs(-0.00005*30/0.03+0.05)<1e-9,"flat engine charge 30 bars: -0.10R (1.5%), -0.05R (3%)")
print("\nFAILS:",fails); sys.exit(1 if fails else 0)
