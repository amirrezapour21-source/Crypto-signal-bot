name: Read-only N2-E development

on:
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: readonly-n2-dev
  cancel-in-progress: false

jobs:
  dev:
    runs-on: ubuntu-latest
    timeout-minutes: 60
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: "3.11"
      - run: pip install pandas numpy requests
      - name: N2-E development run (read-only)
        run: python n2_exec_edge_dev.py
